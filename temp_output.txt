import React from 'react';

/**
 * Button - A customizable button component aligned with Archon UI Design System (3737)
 *
 * Implements the neon/glassmorphism style from the main design system.
 */
export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'outline' | 'ghost' | 'danger';
  size?: 'xs' | 'sm' | 'md' | 'lg';
  accentColor?: 'purple' | 'green' | 'pink' | 'blue' | 'cyan' | 'orange' | 'indigo';
  neonLine?: boolean;
  icon?: React.ReactNode;
  isLoading?: boolean;
}

export const Button: React.FC<ButtonProps> = ({
  children,
  variant = 'primary',
  size = 'md',
  accentColor = 'indigo', // Default to indigo to match current theme, map to 'purple' if needed
  neonLine = false,
  icon,
  isLoading = false,
  className = '',
  disabled,
  ...props
}) => {

  // Size variations
  const sizeClasses = {
    xs: 'text-xs px-2 py-1 rounded',
    sm: 'text-xs px-3 py-1.5 rounded-md',
    md: 'text-sm px-4 py-2 rounded-lg',
    lg: 'text-base px-6 py-2.5 rounded-lg'
  };

  // Map 'indigo' to 'purple' style if needed, or handle custom indigo classes
  // For simplicity, we'll assume the Tailwind config supports these colors or use arbitrary values
  const getColorClass = (color: string) => {
      // Basic mapping for tailwind classes availability
      const map: Record<string, string> = {
          'purple': 'purple',
          'indigo': 'indigo', // archon-ui uses purple as primary, but enduser uses indigo
          'green': 'emerald',
          'blue': 'blue',
          'red': 'red',
          'orange': 'orange',
          'pink': 'pink',
          'cyan': 'cyan'
      };
      return map[color] || color;
  };

  const c = getColorClass(accentColor);

  // Style variations based on variant
  const variantClasses = {
    primary: `
      relative overflow-hidden font-medium
      bg-${c}-600 text-white
      border border-${c}-500/50 border-t-${c}-400
      shadow-md shadow-${c}-500/20 hover:shadow-lg hover:shadow-${c}-500/30 hover:bg-${c}-700
      active:scale-95 transition-all duration-200
      disabled:opacity-50 disabled:cursor-not-allowed
      group
          focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-${c}-500 focus-visible:ring-offset-2
    `,
    // For Archon UI alignment, 'secondary' is often dark glass or solid neutral
    secondary: `
      bg-white text-gray-700 border border-gray-200
      hover:bg-gray-50 hover:border-gray-300 hover:text-gray-900
      active:bg-gray-100 active:scale-95 transition-all duration-200
      disabled:opacity-50 disabled:cursor-not-allowed
          focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gray-400 focus-visible:ring-offset-2
    `,
    outline: `
      bg-transparent border border-${c}-500 text-${c}-600
      hover:bg-${c}-50 active:bg-${c}-100
      disabled:opacity-50 disabled:cursor-not-allowed transition-all
          focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-${c}-500 focus-visible:ring-offset-2
    `,
    ghost: `
      bg-transparent text-gray-600 hover:bg-gray-100 hover:text-gray-900
      disabled:opacity-50 disabled:cursor-not-allowed transition-all
          focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gray-400 focus-visible:ring-offset-2
    `,
    danger: `
      bg-red-600 text-white border border-red-500
      hover:bg-red-700 shadow-md shadow-red-500/20
      active:scale-95 transition-all
          focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-red-500 focus-visible:ring-offset-2
    `
  };

  return (
    <button
      disabled={disabled || isLoading}
      className={`
        inline-flex items-center justify-center
        ${variantClasses[variant]}
        ${sizeClasses[size]}
        ${className}
      `}
      {...props}
    >
      {/* Loading Spinner */}
      {isLoading && (
        <svg className="animate-spin -ml-1 mr-2 h-4 w-4 text-current" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
          <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
          <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
        </svg>
      )}

      {/* Content with icon support */}
      <span className="relative z-10 flex items-center justify-center gap-2">
        {icon && <span>{icon}</span>}
        {children}
      </span>

      {/* Optional neon line (Archon UI signature) */}
      {neonLine && variant === 'primary' && (
        <span className={`absolute bottom-0 left-[15%] right-[15%] w-[70%] mx-auto h-[2px] bg-${c}-400/50 blur-[1px]`}></span>
      )}
    </button>
  );
};
